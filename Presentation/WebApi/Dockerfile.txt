# /Presentation/WebApi/Dockerfile (Bu dosya bu klasörde)

# Stage 1: Build Stage
FROM mcr.microsoft.com/dotnet/sdk:8.0 AS build
WORKDIR /src # Konteyner içinde /src dizinine geçiyoruz

# Çözüm dosyasını ve WebApi proje dosyasını kopyala
# Dockerfile'ın bulunduğu yerden (Presentation/WebApi) çözüm dosyasına ulaşmak için '../../' kullanmalıyız.
# WebApi projesi için 'Presentation/WebApi/Presentation.WebApi.csproj' yolu Dockerfile'a göre doğru değildir.
# Çünkü Dockerfile zaten 'Presentation/WebApi' klasörünün içinde.
# Bu yüzden, kendi klasöründeki projeyi kopyalarken sadece proje dosyasının adını kullanmalıyız.

# Çözüm dosyasını /src dizinine kopyala
COPY "../../ArizaTakipSistemiWithAI.sln", "./"

# WebApi projesi ve varsa diğer bağımlı projelerin .csproj dosyalarını kopyala
# Dockerfile'ın kendisi WebApi klasöründe olduğu için, kaynak yolu './' ile başlar.
COPY "Presentation.WebApi.csproj", "Presentation/WebApi/"

# Diğer bağımlı projeler varsa (örneğin Application, Domain, Infrastructure)
# Dockerfile'ın bulunduğu yerden (Presentation/WebApi) diğer projelere ulaşmak için '../' veya '../../' kullanılır.
# Örneğin, Application klasörü Solution'ın altında ve WebApi ile aynı seviyede değilse:
COPY "../../../Application/Application.csproj", "Application/"
COPY "../../../Domain/Domain.csproj", "Domain/"
COPY "../../../Infrastructure/Infrastructure.csproj", "Infrastructure/"


# Bağımlılıkları geri yükle
# Restore komutu çözüm dosyasının yolunu veya WebApi projesinin tam yolunu bekler (kopyalandığı yere göre).
RUN dotnet restore "./Presentation.WebApi/Presentation.WebApi.csproj"

# Tüm proje dosyalarını kopyala
# Şimdi Dockerfile'ın bulunduğu dizindeki tüm içeriği /src/Presentation/WebApi'e kopyalıyoruz.
# Kaynak yolu '.' Dockerfile'ın bulunduğu dizini (WebApi) temsil eder.
# Hedef yolu 'Presentation/WebApi/' de /src altında bir klasör oluşturur ve oraya kopyalar.
COPY . "Presentation/WebApi/"

# WebApi proje klasörüne geç
WORKDIR "/src/Presentation/WebApi"

# Uygulamayı derle ve yayınla
RUN dotnet publish -c Release -o /app/publish --no-restore

# Stage 2: Runtime Stage
FROM mcr.microsoft.com/dotnet/aspnet:8.0 AS final
WORKDIR /app

COPY --from=build /app/publish .

ENV ASPNETCORE_URLS="http://+:80"
EXPOSE 80

ENTRYPOINT ["dotnet", "Presentation.WebApi.dll"]
